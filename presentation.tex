\subsection{Modèle de puissance par torseurs}

\begin{frame}
    \assertion{La puissance de coupe s'obtient par le comoment du torseur d'action et du torseur cinématique à la pointe de la dent}
    \vspace{-5mm}
    \begin{align*}
        P_{\text{Outil/Pièce}} = & \{\tau_{\text{Outil}\to\text{Pièce}}\} \otimes \{\mathcal{V}_{\text{Pièce/Outil}}\}                                                                                                             \\
                               = & \uncover<3->{\alert<3>{- F_{\vec{e}_r,\text{Outil}\to\text{Pièce}} \|V_{F,\vec{x},\text{Outil/Pièce}}\| \cos(\theta)}} \uncover<2->{\alert<2>{- M_{P,\vec{e}_z,\text{Outil}\to\text{Pièce}} \|\omega_{\vec{z},\text{Outil/Pièce}}\|}} \\
                               & \alert<1>{- F_{\vec{e}_\theta,\text{Outil}\to\text{Pièce}} \left(\|\omega_{\vec{z},\text{Outil/Pièce}}\|\left(R+\frac{a_p}{2}\cot(\kappa_r)\right)-\|V_{F,\vec{x},\text{Outil/Pièce}}\| \sin(\theta)\right)}
    \end{align*}

    \begin{columns}[T]
        \column{0.45\linewidth}
        \begin{figure}
            \centering
            \def\svgwidth{\textwidth}
            \import{images/}{schema_fraise.pdf_tex}
            \caption{Repères et notations}
        \end{figure}

        \column{0.5\linewidth}
        \begin{itemize}
            \item terme de \alert<1>{coupe} ($F_{\vec{e}_\theta,\text{Outil}\to\text{Pièce}}$) : prépondérant

\subsection{Modèle empirique et algorithme RPEM}

\begin{frame}
    \assertion{Expérimentalement, $W_c(V_c)$ présente bien un minimum : un modèle parabolique suffit à le capturer}

    \begin{columns}[c]
        \column{0.55\linewidth}
        \begin{figure}
            \centering
            {\fontsize{10pt}{5pt}\selectfont
                \def\svgwidth{\textwidth}
                \import{images/}{energie_spec_Wc.pdf_tex}}
            \caption{Énergie spécifique $W_c$ en fonction de la vitesse de coupe $V_c$}
        \end{figure}
        \column{0.4\linewidth}
        \begin{alertblock}{Rappel : pourquoi $W_c$ ?}
            \vspace{-1mm}
            \[
                W_c = \frac{\mathcal{P}_{\text{coupe}}}{Q} \quad \text{(J/cm}^3\text{)}
            \]
            \vspace{-2mm}
            {
            \begin{itemize}
                \item Énergie pour enlever \alert{1\,cm$^3$} de matière
                \item Minimiser $\mathcal{P}$ seule revient à ralentir ; ramenée au débit $Q$, elle compare des conditions à \alert{productivité différente}
                \item Critère du couple outil-matière (NF~E66-520-6)
            \end{itemize}}
        \end{alertblock}
        \begin{block}{Modèle empirique parabolique}
            \vspace{-2mm}
            \[
                \hat W_c = a_w V_c^2 + b_w V_c + c_w
            \]
        \end{block}
        \begin{exampleblock}{Minimum analytique}
            \vspace{-2mm}
            \[
                V_{c,\text{opt}} = -\frac{b_w}{2a_w} \quad (a_w > 0)
            \]
        \end{exampleblock}
    \end{columns}
    % SOURCE : fig. 4.14 ; éq. (4.13)-(4.14).
    % ORAL : rappel W_c en 20 s : la puissance seule croît avec V_c (plus de débit),
    % donc la minimiser ferait juste ralentir ; W_c = énergie par cm^3, indépendante de la productivité.
    % ORAL : l'énergie par tour ne fonctionne pas (fig. 4.13), minimum noyé dans le bruit.
\end{frame}

\begin{frame}
    \assertion{La RPEM ré-identifie le modèle à chaque itération et suit l'optimum pendant l'usinage}

    \begin{columns}[c]
        \column{0.35\linewidth}
        {\small
        \begin{algorithmic}[1]
            \Require $V_{c,\text{init}}$, $f_z$, $\hat{\theta}_0$, $R_0$, $\gamma_0$
            \Do
                \State $y_k \gets \bigl[W_c,\ F_r,\ F_\theta,\ M_z\bigr]^\top$ \hfill {\footnotesize\color{jvgris}mesure}
                \State $\hat{y}_k \gets \bigl[a_w V_c^2 + b_w V_c + c_w,\ K f_z a_p\bigr]$
                \State $\varepsilon_k = y_k - \hat{y}_k$
                \State $R_k = R_{k-1} + \gamma_k\bigl(\psi_k^\top \psi_k - R_{k-1}\bigr)$
                \State $\hat{\theta}_k = \hat{\theta}_{k-1} + \gamma_k R_k^{-1} \psi_k^\top \varepsilon_k$ \hfill {\footnotesize\color{jvgris}modèle}
                \State \alert{$V_c^\star = -b_w / 2a_w$} \hfill {\footnotesize\color{jvgris}optimum}
                \State $V_c \gets \mathrm{sat}\bigl(V_c^\star\bigr)$ \hfill {\footnotesize\color{jvgris}pas $h_{V_c}$, bornes}
            \doWhile{usinage en cours}
        \end{algorithmic}}
        \vspace{1mm}
        {\footnotesize\color{jvgris} $\hat{\theta} = [a_w,\ b_w,\ c_w,\ K_f,\ K_c,\ K_m]^\top$, \quad $K \in \{K_f, K_c, K_m\}$\par}
        \column{0.65\linewidth}
        \begin{figure}
            \centering
            \resizebox{\linewidth}{!}{\input{images/schema_rpem.tex}}
            \caption{Schéma de la boucle RPEM}
        \end{figure}

    \end{columns}

\end{frame}

\section{Puissance par comoment}
\begin{frame}
    % Éq. (2.18) complète, avec les trois termes colorés.
    \begin{equation*}
        \left\{\tau_{\text{Outil}\to\text{Pièce}}\right\} = \prescript{}{P}{
            \left\{
            \begin{array}{cc}
                F_{\vec{e}_r,\text{Outil}\to\text{Pièce}}      & M_{P,\vec{e}_r,\text{Outil}\to\text{Pièce}}      \\
                F_{\vec{e}_\theta,\text{Outil}\to\text{Pièce}} & M_{P,\vec{e}_\theta,\text{Outil}\to\text{Pièce}} \\
                F_{\vec{e}_z,\text{Outil}\to\text{Pièce}}      & M_{P,\vec{e}_z,\text{Outil}\to\text{Pièce}}      \\
            \end{array}
            \right\}_{(\vec{e}_r,\vec{e}_\theta,\vec{e}_z)}}
    \end{equation*}
    \begin{align*}
        \left\{\mathcal{V}_{\text{Pièce}/\text{Outil}}\right\} & = \prescript{}{F}{
            \left\{
            \begin{array}{cc}
                0                                               & \|V_{F,\vec{x},\text{Outil}/\text{Pièce}}\| \\
                0                                               & 0                                           \\
                -\|\omega_{\vec{z},\text{Outil}/\text{Pièce}}\| & 0
            \end{array}
        \right\}_{(\vec{x},\vec{y},\vec{z})}}                                       \\
                                                               & = \prescript{}{P}{
            \left\{
            \begin{array}{cc}
                0                                              & \|V_{F,\vec{x},\text{Outil}/\text{Pièce}}\|\cos(\theta)                                                                                             \\
                0                                              & \|\omega_{\vec{z},\text{Outil}/\text{Pièce}}\| \left(R+\frac{a_p}{2}\cot(\kappa_r)\right) - \|V_{F,\vec{x},\text{Outil}/\text{Pièce}}\|\sin(\theta) \\
                \|\omega_{\vec{z},\text{Outil}/\text{Pièce}}\| & 0
            \end{array}
            \right\}_{(\vec{e}_r,\vec{e}_\theta,\vec{e}_z)}}
    \end{align*}
    \begin{equation*}
        \begin{split}
            \mathcal{P}_{\text{Outil}/\text{Pièce}} = & \left\{\tau_{\text{Outil}\to\text{Pièce}}\right\} \otimes \left\{\mathcal{V}_{\text{Pièce}/\text{Outil}}\right\}                                                                                                        \\
            =                                         & -F_{\vec{e}_r,\text{Outil}\to\text{Pièce}} \cdot \|V_{F,\vec{x},\text{Outil}/\text{Pièce}}\|\cos(\theta)                                                                                                                \\
                                                      & - F_{\vec{e}_\theta,\text{Outil}\to\text{Pièce}} \cdot \left(\|\omega_{\vec{z},\text{Outil}/\text{Pièce}}\| \left(R+\frac{a_p}{2}\cot(\kappa_r)\right) - \|V_{F,\vec{x},\text{Outil}/\text{Pièce}}\|\sin(\theta)\right) \\
                                                      & - M_{P,\vec{e}_z,\text{Outil}\to\text{Pièce}} \cdot \|\omega_{\vec{z},\text{Outil}/\text{Pièce}}\|.
        \end{split}
    \end{equation*}
\end{frame}

\begin{frame}
    \frametitle{Algorithme CLRIVFR détaillé}
    {\footnotesize
    \begin{algorithmic}[1]
        \Require $y$, $r_1$, $r_2$, $C_P$, $C_I$, $P\!F$, $T_e$, $\omega_c$, $n_a$, $n_b$, $n_k$
        \State Initialisation de $\hat{\theta}_{riv}^0$ par CLSRIVFR (ou valeurs initiales)
        \State Signal d'excitation $r$ à partir de $r_1$, $r_2$, $C_P$, $C_I$ et du préfiltre $P\!F$
        \State Commande $u$ à partir de $r$, de la mesure $y$ et des régulateurs $C_P$, $C_I$
        \State $\text{iter} \gets 0$
        \Do
            \State $\text{iter} \gets \text{iter} + 1$
            \State Mise à jour du modèle : $\hat{G}^{\text{iter}}(q) = B(q,\hat{\theta}^{\text{iter}-1}) / A(q,\hat{\theta}^{\text{iter}-1})$
            \State Commande instrumentale $\nu$ et sortie instrumentale $x$ (système auxiliaire sans bruit)
            \State Bruit coloré estimé : $\hat{\xi}(k) = \hat{S}\bigl(q,\hat{\theta}_{riv}^{\text{iter}-1}\bigr)^{-1}\bigl(y(k) - \hat{x}(k)\bigr)$
            \State Modèle de bruit $H(q,\hat{\eta}_n)$ estimé sur $\hat{\xi}$ (fonction \texttt{armax} de Matlab)
            \State Conception du filtre optimal $F'_{opt}(q)$
            \State Filtrage de $y$, $u$, $x$, $\nu$ par $F'_{opt}(q)$ $\Rightarrow$ $y_f$, $u_f$, $x_f$, $\nu_f$
            \State Vecteur de régression $\varphi_f$ (à partir de $y_f$, $u_f$) et vecteur instrumental $z_f$ (à partir de $x_f$, $\nu_f$)
            \State Mise sous forme matricielle : $\Phi_f$, $Y_f$, $Z_f$
            \State Solution des variables instrumentales : $\hat{\theta}_{riv} = \left(Z_f^\top \Phi_f\right)^{-1} Z_f^\top Y_f$
        \doWhile{$\displaystyle\sum_{j=1}^{\dim \hat{\theta}}\left|\frac{\hat{\theta}_{riv,j}^{\text{iter}}-\hat{\theta}_{riv,j}^{\text{iter}-1}}{\hat{\theta}_{riv,j}^{\text{iter}}}\right| > \epsilon$}
        \Ensure $\hat{\theta}_{riv}$
    \end{algorithmic}}
\end{frame}
