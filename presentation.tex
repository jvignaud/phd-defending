\subsection{Modèle de puissance par torseurs}

\begin{frame}
    \assertion{La puissance de coupe s'obtient par le comoment du torseur d'action et du torseur cinématique à la pointe de la dent}
    \vspace{-5mm}
    \begin{align*}
        P_{\text{Outil/Pièce}} & = \{\tau_{\text{Outil}\to\text{Pièce}}\} \otimes \{\mathcal{V}_{\text{Pièce/Outil}}\}                                                                                                             \\
                               & = - F_{\vec{e}_r,\text{Outil}\to\text{Pièce}} \|V_{F,\vec{x},\text{Outil/Pièce}}\| \cos(\theta) - M_{P,\vec{e}_z,\text{Outil}\to\text{Pièce}} \|\omega_{\vec{z},\text{Outil/Pièce}}\|             \\
                               & - F_{\vec{e}_\theta,\text{Outil}\to\text{Pièce}} \left(\|\omega_{\vec{z},\text{Outil/Pièce}}\|\left(R+\frac{a_p}{2}\cot(\kappa_r)\right)-\|V_{F,\vec{x},\text{Outil/Pièce}}\| \sin(\theta)\right)
    \end{align*}

    \begin{columns}[T]
        \column{0.45\linewidth}
        \begin{figure}
            \centering
            \def\svgwidth{\textwidth}
            \import{images/}{schema_fraise.pdf_tex}
            \caption{Repères et notations}
        \end{figure}

        \column{0.5\linewidth}
        \begin{itemize}
            \item terme de \alert{coupe} ($F_{\vec{e}_\theta,\text{Outil}\to\text{Pièce}}$) : prépondérant
            \item terme de \alert{moment} ($M_{P,\vec{e}_z,\text{Outil}\to\text{Pièce}}$) : non négligeable
            \item terme d'avance ($F_{\vec{e}_r,\text{Outil}\to\text{Pièce}}$) : négligeable
        \end{itemize}
    \end{columns}
    % SOURCE : fig. 2.1 / 2.6 ; éq. (2.18) en annexe si question.

            \footnotesize
            \begin{itemize}
                \item $V_c$ est imposée par la \alert{broche}, asservie : il faut connaître sa dynamique (section 4)
                \item $W_c(V_c)$ doit être \alert{identifiée en ligne} (section 5)
            \end{itemize}
        \end{block}

        \column{0.45\linewidth}
        \begin{itemize}
            \item \textbf{Pourquoi $W_c$ ?}
                  \begin{itemize}
                      \item énergie nécessaire pour enlever 1~cm$^3$ de matière
                      \item critère de référence du couple outil-matière (NF~E66-520-6)
                      \item ramenée au débit : compare des conditions de coupe à productivité différente
                      \item objectif : \alert{minimiser} $W_c(V_c, f_z)$
                  \end{itemize}
        \end{itemize}
        \begin{equation*}
            W_c = \frac{P_{\text{méca}}}{Q}, \qquad Q = a_e\, a_p\, V_f
        \end{equation*}
        \begin{itemize}
            \item Modèle statique : $V_c$ n'intervient pas, alors que la norme COM observe une évolution de $W_c$ avec $V_c$
        \end{itemize}
    \end{columns}
    % SOURCE : fig. 2.16 ; éq. (2.31)-(2.32).

\end{frame}

\begin{frame}
    \assertion{Expérimentalement, $W_c(V_c)$ présente bien un minimum : un modèle parabolique suffit à le capturer}

    \begin{columns}[c]
        \column{0.55\linewidth}
        \begin{figure}
            \centering

        \begin{block}{Modèle empirique parabolique}
            \vspace{-2mm}
            \[
                \hat W_c = a_w V_c^2 + b_w V_c + c_w

            \vspace{-2mm}

        \end{exampleblock}
    \end{columns}
