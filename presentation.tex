\subsection{Contexte industriel et enjeux}

\begin{frame}
    \assertion{Chez VLM Robotics, chaque pièce est unique : impossible de régler les conditions de coupe par essais successifs}

    \begin{columns}[c]
        \column{0.5\linewidth}
        \begin{figure}
            \centering
            \includegraphics[width=\linewidth]{images/GEMINI_CEA.png}
            \caption{Cellule Gemini de fabrication et d'usinage de pièce XXL}
        \end{figure}
        \column{0.45\linewidth}
        \begin{itemize}
            \item Industrie 4.0 : production unitaire ou très petites séries, à cadence industrielle
            \item Fabrication additive (WAAM), usinage (fraisage, tournage, perçage), CND
            \item Clients aéronautique, spatial, défense
            \item Pas d'effet de série $\Rightarrow$ pas de campagne de réglage par référence
            \item Besoin d'une stratégie en ligne, pièce par pièce, avec instrumentation simple, non invasif et transposable d'un poste à l'autre
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}
    \assertion{Maîtriser l'énergie mécanique de coupe, c'est maîtriser la tenue en fatigue de la pièce et réduire la consommation}

    \begin{columns}[c]
        \column{0.5\linewidth}
        \begin{figure}
            \centering
            \begin{subfigure}{\linewidth}
                \centering
                \includegraphics[width=\linewidth]{images/CR_meca.png}
                \caption{Charge mécanique}
            \end{subfigure} \\
            \begin{subfigure}{\linewidth}
                \centering
                \includegraphics[width=\linewidth]{images/CR_thermique.png}
                \caption{Charge thermique}
            \end{subfigure}
            \caption{Phases de formation des contraintes résiduelles dans la pièce}
        \end{figure}
        \column{0.45\linewidth}
        \begin{itemize}
            \item Thermique dominant $\Rightarrow$ contraintes résiduelles de \alert{traction} en surface
            \item Mécanique dominant $\Rightarrow$ contraintes de \alert{compression}, favorables à la fatigue
            \item Double enjeu : métallurgique et énergétique
        \end{itemize}
    \end{columns}
    % SOURCE : Introduction, § Contexte.
\end{frame}

    \begin{itemize}
        \item La boucle de vitesse de la broche (Sinamics) \alert{ne peut pas être désactivée}
        \item Le modèle doit être \alert{calculable en ligne}
    \end{itemize}
    % ORAL : poser ici la question de recherche en une phrase.
\end{frame}

\begin{frame}
    \assertion{Quatre contributions, de la mesure jusqu'à l'optimisation sur machine}

    \begin{enumerate}
        \item \textbf{Instrumentation} : étalonnage de la platine Kistler par cobot, dépendant du point d'application
        \item \textbf{Modélisation} : formulation générique de la puissance de coupe par torseurs, faisant apparaître le terme de moment ($K_m$), identifiée et confrontée à la puissance électrique broche
        \item \textbf{Identification en boucle fermée} : 4 algorithmes pour PI parallèle + préfiltre sur la seule voie intégrale
        \item \textbf{Optimisation en ligne} : RPEM sur un modèle empirique $W_c(V_c)$
    \end{enumerate}
    % ORAL : annoncer que la contribution 3 est le cœur théorique.
\end{frame}

\subsection{Modèle empirique et algorithme RPEM}

\begin{frame}
    \assertion{Expérimentalement, $W_c(V_c)$ présente bien un minimum : un modèle parabolique suffit à le capturer}

    \begin{columns}[c]
        \column{0.55\linewidth}
        \figafaire{Mesures de $W_c$ en fonction de $V_c$ (médiane)\\+ parabole ajustée, minimum marqué}
        \column{0.4\linewidth}
        \[
            \hat W_c = a_w V_c^2 + b_w V_c + c_w
        \]
        \[
            V_{c,\text{opt}} = -\frac{b_w}{2a_w} \quad (a_w > 0)
        \]
    \end{columns}
    % SOURCE : fig. 4.14 ; éq. (4.13)-(4.14).
    % ORAL : l'énergie par tour ne fonctionne pas (fig. 4.13), minimum noyé dans le bruit.
\end{frame}

\begin{frame}
    \assertion{En avance, $W_c(f_z)$ décroît strictement : cohérent avec le modèle, mais pas de minimum exploitable}

    \figafaire{$W_c$ en fonction de $f_z$, avec la borne préconisée par le constructeur}
    % SOURCE : fig. 4.16. Seul minimum vers 0,42 mm/(tr.dent), hors préconisations.

\begin{frame}
    \assertion{La RPEM ré-identifie le modèle à chaque itération et suit l'optimum pendant l'usinage}
