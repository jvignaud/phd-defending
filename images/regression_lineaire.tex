% Régressions linéaires KT110 (référence) vs platine Kistler, vis 1
% Données : images/data/reg_*.dat (x = KT110, y = Kistler), extraites de l'export MATLAB
% Chaque graphe a les mêmes bornes en x et en y : la droite 1:1 est la diagonale.
% Nécessite \usepgfplotslibrary{groupplots} dans le préambule.
\begin{tikzpicture}
\pgfplotsset{
    mesure/.style={only marks, mark=*, mark size=0.7pt, jvbleu, fill opacity=0.6, draw opacity=0.6},
    regression/.style={thick, jvcorail, samples=2},
    unpourun/.style={thin, dashed, jvgris, samples=2},
}
\begin{groupplot}[
    group style={group size=2 by 1, horizontal sep=11mm},
    width=0.27\linewidth, height=0.27\linewidth,
    scale only axis,
    every axis/.append style={font=\scriptsize},
    tick label style={font=\tiny},
    label style={font=\scriptsize},
    title style={font=\scriptsize\bfseries, text=jvbleu, yshift=-1mm},
    /pgf/number format/use comma, /pgf/number format/1000 sep={},
    grid=major, grid style={jvgris!20},
    axis line style={jvgris},
    tick style={jvgris},
    xlabel={KT110},
    /tikz/eqn/.style={font=\tiny, fill=white, fill opacity=0.85, text opacity=1,
                inner sep=1.2pt, anchor=south east},
    legend style={font=\tiny, draw=none, fill=none, legend columns=3,
                  /tikz/every even column/.append style={column sep=3mm}},
]
    % ---------------------------------------------------------------- Force
    \nextgroupplot[
        title={Force (N)},
        ylabel={Kistler},
        xmin=-51.8, xmax=-45.4, ymin=-51.8, ymax=-45.4,
        xtick={-51,-49,-47}, ytick={-51,-49,-47},
        legend to name=legreg,
    ]
    \addplot[mesure] table[x=x, y=y] {images/data/reg_force.dat};
    \addlegendentry{Mesures}
    \addplot[regression, domain=-51.8:-45.4] {1.00219*x};
    \addlegendentry{Régression linéaire}
    \addplot[unpourun, domain=-51.8:-45.4] {x};
    \addlegendentry{Droite 1:1}
    \node[eqn] at (rel axis cs:0.97,0.03) {$y = 1{,}002\,x$};

    % ---------------------------------------------------------------- Couple
    \nextgroupplot[
        title={Couple (N$\cdot$m)},
        xmin=0.88, xmax=1.05, ymin=0.88, ymax=1.05,
        xtick={0.9,0.95,1}, ytick={0.9,0.95,1},
        xticklabel style={/pgf/number format/fixed, /pgf/number format/precision=2},
        yticklabel style={/pgf/number format/fixed, /pgf/number format/precision=2},
    ]
    \addplot[mesure] table[x=x, y=y] {images/data/reg_couple_pos.dat};
    \addplot[regression, domain=0.88:1.05] {0.977474*x - 0.0313789};
    \addplot[unpourun, domain=0.88:1.05] {x};
    \node[eqn, anchor=north west] at (rel axis cs:0.03,0.97) {$y = 0{,}977\,x - 0{,}031$};
\end{groupplot}
% légende commune, centrée au-dessus des deux graphes
\node[anchor=south] at ([yshift=4mm]$(group c1r1.north)!0.5!(group c2r1.north)$) {\pgfplotslegendfromname{legreg}};
\end{tikzpicture}
